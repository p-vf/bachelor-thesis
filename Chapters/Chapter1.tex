\chapter{Background}

In this chapter, we cover some Definitions relevant for this thesis.

\section{Deterministic Finite Automaton}
\begin{definition}
A deterministic finite automaton (DFA) is a 5-tuple $M = \langle Q, \Sigma, \delta, q_0, F \rangle$ where
\begin{itemize}
    \item $Q$ is the finite set of states
    \item $\Sigma$ is the input alphabet
    \item $\delta : Q \times \Sigma \rightarrow Q$ is the transition function
    \item $q_0 \in Q$ is the start state
    \item $F \subseteq Q$ is the set of accept states (or final states).
\end{itemize}
For readability we define $\delta(q, \varepsilon) = q$ and $\delta(q, aw) = \delta(\delta(q, a), w)$ where $\varepsilon$ denotes the word with length 0, $q \in Q$, $a \in \Sigma$ and $w \in \Sigma^+$.
If the DFA is clear from context, we write $q \cdot w$ instead of $\delta(q, w)$.
\end{definition}

\begin{definition}
DFA $M = \langle Q, \Sigma, \delta, q_0, F \rangle$ accepts the word $w \in \Sigma^*$ if $q \cdot w \in F$.
\end{definition}

\begin{definition}
Let $M$ be a DFA. The language recognized by $M$ is defined as $\mathcal L(M) = \{w \in \Sigma^* \mid w \text{ is accepted by } M\}$.
\end{definition}

\begin{example}
    \label{ex:dfa}
    Let DFA $E = \langle Q_E, \Sigma_E, \delta_E, q_0, F_E \rangle$ where
    \begin{equation*}
        Q_E = {0, 1, 2, 3, 4, 5, 6, 7, 8}, \quad \Sigma_E = \{a, b\}, \quad q_0 = 0, \quad F_E = \{1, 5, 7\},
    \end{equation*}
    and $\delta_E: Q_E \times \Sigma_E \rightarrow Q_E$ is the function defined by
    \begin{align*}
        \delta_E(0, a) &= 1, \quad \delta_E(0, b) = 2, \quad \delta_E(1, a) = 3, \quad \delta_E(1, b) = 4, \quad \delta_E(2, a) = 5, \quad \\
        \delta_E(3, a) &= \delta_E(4, a) = 6, \quad \delta_E(5, a) = \delta_E(5, b) = \delta(6, b) = 7, \quad \\
        \delta_E(2, b) &= \delta_E(3, b) = \delta_E(4, b) = \delta_E(6, a) = \delta_E(7, a) = \delta_E(7, b) = \delta_E(8, a) = \delta_E(8, b) = 8.
    \end{align*}
    One can check that the language recognized by $E$ is $\mathcal L(E) = \{a, b, ba, baa, bab, aaab, abab\}$.
\end{example}

We depict a DFA visually as a directed graph with labelled arcs, where there is a corresponding node for each state and a labelled arc for each transition in the transition function.
States from which no final state is reachable - so called sink states - and corresponding transitions can be omitted in the visual depiction.
An example is given by the visual depiction of DFA $E$ from Example \ref{ex:dfa} in Figure \ref{fig:ex_dfa}.
We see that the state 8 is not depicted because it is a sink state (once the state 8 is reached, no other state can be visited because $\delta_E(8, a) = \delta_E(8, b) = 8$ and $8 \notin F_E$).

\begin{definition}
DFA $M = \langle Q, \Sigma, \delta, q_0, F \rangle$ is acyclic, if $q \cdot w \neq q$ for all words $w \in \Sigma^*$ and states $q \in Q$.
\end{definition}

\section{Planning}

\begin{definition}
    A state space or transition system is a 6-tuple $\mathcal S = \langle S, A, \textit{cost}, T, s_I, S_G \rangle$ with
    \begin{itemize}
        \item finite set of states $S$
        \item finite set of actions $A$
        \item action costs $\textit{cost}: A \rightarrow \mathbb R_0^+$
        \item transition relation $T \subseteq S \times A \times S$ for which $\langle s, a, s' \rangle \in T$ implies $\langle s, a, s'' \rangle \notin T$ for all $s'' \neq s'$ ($T$ is deterministic in $\langle s, a \rangle$)
        \item initial state $s_I \in S$
        \item set of goal states $S_G \subseteq S$
    \end{itemize}
\end{definition}
In the following definitions, let $\mathcal S$ be a state space as above.
\begin{definition}
    The triples $\langle s, a, s' \rangle \in T$ are called transitions. We say $\mathcal S$ has the transition $\langle s, a, s' \rangle$ if $\langle s, a, s' \rangle \in T$ and write this as $s \stackrel{a}{\rightarrow} s'$.
\end{definition}
\begin{definition}
    Let $s_0, \ldots, s_n \in S$ be states and $a_1, \ldots, a_n \in A$ be actions such that $s_0 \stackrel{a_1}{\rightarrow} s_1, \ldots, s_{n-1} \stackrel{a_n}{\rightarrow} s_n$.
    \begin{itemize}
        \item $\pi = \langle a_1, \ldots, a_n \rangle$ is a path from $s_0$ to $s_n$
        \item length of $\pi$ is $\lvert \pi \rvert = n$
        \item cost of $\pi$ is $\textit{cost}(\pi) = \sum_{i=1}^{n} \textit{cost}(a_i)$
    \end{itemize}
\end{definition}
\begin{definition}
    \begin{itemize}
        \item state $s \in S$ is reachable if a path from $s_I$ to $s$ exists
        \item paths from $s \in S$ to some state $s_G \in S_G$ are solutions for/from $s$
        \item solutions for $s_I$ are called solutions for $\mathcal S$
        \item optimal solutions (for $s$) have minimal costs among all solutions (for $s$)
    \end{itemize}
\end{definition}



\chapter{Algorithms for Compressing Sets of Plans}
The result of top-k planners such as SymK (citation needed) is a finite set of plans, i.e. a set of sequences of actions, or in other words a finite language over the actions. In our case, these finite sets of languages are explicitly represented as a set of words. DFAs and DFCAs alike can be used to compactly represent finite languages. In this chapter, we take a look at how we can construct DFAs and DFCAs from the explicit representation of the language that represent the same language, i.e. given a finite language $L \subset A^*$, how do we construct a DFA (or DFCA) $M$ with $\mathcal L(M) = L$.


\begin{definition}
Let $M = \langle Q, \Sigma, \delta, q_0, F \rangle$ be an acyclic DFA. The height $h(q)$ of a state $q \in Q$ is defined as the length of the longest word $w \in \Sigma^*$ for which $q \cdot w \in F$.
\end{definition}

\begin{definition}
Let $M = \langle Q, \Sigma, \delta, q_0, F \rangle$ be a DFA. Then the signature of state $q$ is the set
\[S(q) = \{(a, q') \mid a \in \Sigma, q \cdot a = q'\} \cup \begin{cases}\{+\} &\text{if } q \in F \\\{-\} &\text{otherwise.}\end{cases}\]

Two states are called equivalent iff they have the same signature.
\end{definition}

\section{Generating a DFA Given a Finite Language}
To generate a DFA given a finite set of words, we use the method of constructing a tree-shaped DFA. This DFA is a prefix-tree, also known as a trie.
To construct this DFA, we start with a DFA consisting of only the initial state $q_0$, and a sink state $\bot$, neither of which is a final state. Initially, $q \cdot a = \bot$ holds for any state $q$, also for states that are added during the following procedure.
We then iterate over all words in the language.
For each word, we iterate over its symbols - here labelled $a$ -, keeping track in which state we are in the DFA - here labelled $q$ - until we (a) reach the end of the word, at which point $q$ is marked as a final state or (b) the successor state $q \cdot a = \bot$ is the sink state, at which point we add a new state $q'$ with $q \cdot a = q'$.

We obtain a DFA that recognizes exactly the language that was given. However, notice that this DFA is not very efficient with respect to the number of states: for example, all final states without successor could be merged into one state without changing the language.
To address this inefficiency, Revuz's Algorithm for minimizing DFAs can be utilized, which is described in the following section.

\section{Merging States}
During Revuz's algorithm, the $\textit{merge}$ operation is performed repeatedly.
This operation is defined by $A' = \textit{merge}(A, q, q')$ where $A = \langle Q_A, \Sigma, \delta_A, q_0, F_A \rangle$ is the original DFA and $q$, $q'$ are states in $Q_A$ and $A' = \langle Q_A \setminus q', \Sigma, \delta_{A'}, q_0, F_A \setminus q' \rangle$ is the resulting DFA with
\[\delta_{A'}(q, a) = \begin{cases}
    \delta_A(p, a), &\text{if } \delta_A(p, a) \ne q' \\
    q, &\text{otherwise.}
\end{cases}\]
This operation only makes sense if the two states $q, q'$ are equivalent. An example application of this operation is depicted in Figure \ref{fig:ex-merge-dfa}.

\section{Revuz's Algorithm}
Revuz's algorithm finds, given a DFA $M$ with states $Q$ and final states $F$, the minimal DFA $M'$ (with respect to the number of states) for which $\mathcal L(M) = \mathcal L(M')$.

The algorithm works by partitioning the set of states $Q$ into sets $Q_i$ where each state in $Q_i$ has height $i$.
Then, for each $i \in \{0, 1, \ldots\}$ in increasing order we merge all states $q, q' \in Q_i$ for which $S(q) = S(q')$. The result of this is DFA $M'$.

To compute the partition on $Q_i$ we must traverse the DFA from its final states backwards, on each state keeping track of the maximal height. States which are not reached by this traversal are irrelevant and don't have to be included in the partition.

To merge the equivalent states in $Q_i$, we can create a list of the states and sort them lexicographically by their signatures. Equivalent states in the sorted list are next to eachother, so we can merge consecutive equivalent states in this list and be sure that we merged all equivalent states.
